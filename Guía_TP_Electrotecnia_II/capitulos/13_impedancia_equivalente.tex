\info{
Impedancia Equivalente}{
\subsubsection{Paralelo} 
\emph{La impedancia equivalente es la resultante las impedancias individuales, el valor total de los impedancia en paralelo es igual al recíproco de la suma de los recíprocos de las impedancias individuales.}

\begin{equation*}
Z_T= \frac{1}{\frac{1}{Z_1}+\frac{1}{Z_2}} \Rightarrow
Z_T= \frac{1}{\frac{1}{Z_1}+\frac{1}{Z_2}+\dots+\frac{1}{Z_n}}
\end{equation*}

\begin{equation*}
    Z_T=\dfrac{1}{\sum_{i=1}^{n}{\frac{1}{Z_i}}}
\end{equation*}


\subsubsection{Serie}
\emph{La impedancia equivalente de impedancias conectadas en serie es igual a la suma de las impedancias individuales.}

\begin{equation*}
Z_T= Z_1+Z_2 \Rightarrow Z_T= Z_1+Z_2+\dots+Z_n
\end{equation*}

\begin{equation*}
Z_T= \sum_{i=1}^n Z_i
\end{equation*}

Equivalencias: La unidad para impedancia es el Ohm $\Omega$.

}
