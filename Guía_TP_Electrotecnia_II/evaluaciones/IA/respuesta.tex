\section*{Resolución – Evaluación A}

\begin{enumerate}
\item $F = B \cdot I \cdot l = 0{,}8 \cdot 12 \cdot 0{,}2 = \boxed{1{,}92\,N}$

\item $B = 0{,}12\,T,\ l = 0{,}15\,m,\ F = 0{,}45\,N \Rightarrow I = \frac{F}{B \cdot l} = \frac{0{,}45}{0{,}12 \cdot 0{,}15} = \boxed{25\,A}$

\item $L = \frac{N \cdot \Phi}{I} = \frac{400 \cdot 1{,}2 \times 10^{-3}}{30} = \boxed{0{,}016\,H}$

\item $e = -L \cdot \frac{di}{dt} = -0{,}006 \cdot 0{,}2 = \boxed{-1{,}2 \times 10^{-3}\,V}$

\item $\mu_0 = 4\pi \cdot 10^{-7},\ r = 0{,}05,\ l = 0{,}3,\ N = 1500$\\
$A = \pi r^2 = 7{,}85 \times 10^{-3}\,m^2$\\
$L = \frac{\mu_0 N^2 A}{l} = \frac{4\pi \cdot 10^{-7} \cdot 1500^2 \cdot 7{,}85 \cdot 10^{-3}}{0{,}3} \approx \boxed{0{,}074\,H}$

\item $F = B \cdot I \cdot l = 0{,}9 \cdot 10 \cdot 0{,}25 = \boxed{2{,}25\,N}$

\item $e = -L \cdot \frac{di}{dt} = -0{,}01 \cdot (-0{,}5) = \boxed{0{,}005\,V}$

\item Corriente al este, campo hacia abajo: fuerza al norte. (Regla de Fleming izquierda)

\end{enumerate}

\section*{Resolución – Evaluación B}

\begin{enumerate}
\item $F = B \cdot I \cdot l = 1{,}1 \cdot 8 \cdot 0{,}3 = \boxed{2{,}64\,N}$

\item $F = 3{,}5\,N,\ l = 0{,}15\,m,\ I = 18\,A \Rightarrow B = \frac{F}{I \cdot l} = \frac{3{,}5}{18 \cdot 0{,}15} \approx \boxed{1{,}30\,T}$

\item $A = \pi r^2 = \pi \cdot 0{,}1^2 = 0{,}0314\,m^2$\\
$L = \frac{\mu N^2 A}{l} = \frac{2{,}5 \cdot 10^{-6} \cdot 1000^2 \cdot 0{,}0314}{0{,}2} = \boxed{0{,}392\,H}$

\item $e = -L \cdot \frac{di}{dt} = -0{,}005 \cdot 0{,}6 = \boxed{-0{,}003\,V}$

\item $A = \pi r^2 = \pi \cdot 0{,}04^2 = 5{,}026 \cdot 10^{-3}\,m^2$\\
$\mu = \frac{L \cdot l}{N^2 \cdot A} = \frac{0{,}8 \cdot 0{,}15}{1200^2 \cdot 5{,}026 \cdot 10^{-3}} \approx \boxed{2{,}07 \cdot 10^{-5}\,Tm/A}$

\item $F = B \cdot I \cdot l = 0{,}6 \cdot 5 \cdot 0{,}22 = \boxed{0{,}66\,N}$

\item $e = L \cdot \frac{di}{dt} = 0{,}3 = 0{,}02 \cdot \frac{di}{dt} \Rightarrow \frac{di}{dt} = \boxed{15\,A/s}$

\item Corriente al sur, campo de oeste a este ⇒ fuerza hacia abajo (regla de la mano izquierda)

\end{enumerate}
