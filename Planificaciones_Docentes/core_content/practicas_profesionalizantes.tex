% --- OBJETIVOS ---
\section*{Objetivos}
\begin{itemize}[label=\textbullet]
    \item \textbf{General:} Desarrollar competencias profesionales mediante la planificación, ejecución y control de un proyecto técnico real o simulado.
    \item \textbf{Específicos:}
    \begin{itemize}
        \item Implementar herramientas de gestión ágil (Kanban, Gantt) para la organización de tareas.
        \item Realizar estudios de costos, mercado y localización para la viabilidad de un proyecto.
        \item Aplicar normativas legales, marco jurídico y estándares de control de calidad.
        \item Diseñar planes de mantenimiento preventivo y optimizar procesos mediante diagramas de flujo y estudio de tiempos.
    \end{itemize}
\end{itemize}

% --- METODOLOGÍA ---
\section*{Metodología}
\begin{itemize}[label=\textbullet]
    \item \textbf{Enfoque:} Aprendizaje Basado en Proyectos (ABP). El alumno transita desde la idea conceptual hasta la planificación técnica y económica.
    \item \textbf{Estrategias:}
    \begin{itemize}
        \item Desarrollo de un proyecto profesionalizante individual o grupal.
        \item Uso de herramientas de gestión industrial y administrativa.
        \item Investigación de campo y análisis de mercado.
        \item Mentoría técnica sobre la implementación real del proyecto.
    \end{itemize}
\end{itemize}

% --- EVALUACIÓN ---
\section*{Evaluación}
\begin{itemize}[label=\textbullet]
    \item \textbf{Criterios:}
    \begin{itemize}
        \item Calidad técnica y coherencia de los entregables del proyecto.
        \item Capacidad de organización y cumplimiento de los plazos establecidos en el Gantt.
        \item Rigurosidad en el análisis de costos y marco jurídico.
        \item Capacidad de defensa oral y argumentación del proyecto.
    \end{itemize}
    \item \textbf{Instrumentos:}
    \begin{itemize}
        \item Portfolio de avances del proyecto.
        \item Entregas parciales de los estudios técnicos.
        \item Presentación final del proyecto profesionalizante.
    \end{itemize}
\end{itemize}
