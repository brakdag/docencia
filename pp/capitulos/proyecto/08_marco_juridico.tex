\chapter{Aplicación del Marco Jurídico al Proyecto}
\actividad{ Actividad para el proyecto}
\begin{enumerate}
  \item  Inventar una marca para su proyecto y verificar si ya está
    registrada en el INPI (\emph{Instituto Nacional de la Propiedad Industrial})

    \web{https://portaltramites.inpi.gob.ar/}{INPI}

  \item Verificar en el sitio, si hay algún registro de patente del
    producto, realizar un informe de esto, y agregarlo a la carpeta de proyecto.

  \item Inventar una dirección web de o página para su proyecto ej.
    proyecto.com.ar, y verificar si está disponible en NIC.AR
    \web{https://nic.ar/}{NIC AR}
\end{enumerate}
