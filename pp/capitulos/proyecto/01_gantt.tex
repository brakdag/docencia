\chapter{Aplicación de Gantt al Proyecto}
\actividad{Actividad en proyecto: Planificación Temporal}
\begin{enumerate}
  \item Definir la lista completa de tareas necesarias para la ejecución de su proyecto profesionalizante.
  \item Establecer la duración estimada de cada tarea.
  \item Determinar las dependencias entre tareas (cuál debe terminar para que otra comience).
  \item Elaborar el Diagrama de Gantt del proyecto utilizando una herramienta digital (Excel, Project, GanttProject, etc.).
  \item Identificar el camino crítico del proyecto.
\end{enumerate}
